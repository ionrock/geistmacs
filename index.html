Install & Configure

Add Geistmacs to your Emacs setup in seconds.

init.el 
;; -------------------------------------------------------------
;; Vercel Geist Theme & Typography Setup
;; -------------------------------------------------------------
(add-to-list 'custom-theme-load-path (expand-file-name "~/projects/geistmacs"))
(add-to-list 'load-path (expand-file-name "~/projects/geistmacs"))

(use-package geist
  :ensure nil
  :demand t
  :init
  ;; Default font for all GUI frames
  (add-to-list 'default-frame-alist '(font . "Geist Mono-12"))
  :config
  ;; Apply 12pt Geist Mono and 12pt Geist
  (when (display-graphic-p)
    (geist-setup-fonts 12 12))
  ;; Load your preferred default
  (load-theme 'geist-light t)) ; or (load-theme 'geist-dark t)
